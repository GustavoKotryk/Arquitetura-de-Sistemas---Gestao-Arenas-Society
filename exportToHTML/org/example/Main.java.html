package org.example;

import org.example.controller.ArenaController;
import org.example.controller.ReservaController;
import org.example.infra.notifications.WhatsAppService;
import org.example.infra.repository.MemoryArenaRepository;
import org.example.infra.repository.SqlReservaRepository;
import org.example.infra.strategies.PrecoDiaComumStrategy;
import org.example.infra.strategies.PrecoHorarioPicoStrategy;
import org.example.service.ArenaService;
import org.example.service.ReservaService;
import org.example.view.ReservaView;

/**
 * Ponto de entrada — responsável pelo wiring (composição de dependências).
 *
 * Fluxo MVC completo:
 *   Main (wiring)
 *     → View recebe Controllers
 *     → Controllers recebem Services
 *     → Services recebem Repositories + Strategies + Notificações
 *
 * Para trocar banco de dados: substitua SqlReservaRepository por outra implementação de IReservaRepository.
 * Para trocar notificação: substitua WhatsAppService por outra implementação de INotificacaoService.
 * Para trocar estratégia de preço: injete outra implementação de ICalculoPrecoStrategy.
 */
public class Main {
    public static void main(String[] args) {

        // --- Infraestrutura ---
        var reservaRepository = new SqlReservaRepository();       // troque por MemoryReservaRepository para testes
        var arenaRepository   = new MemoryArenaRepository();
        var notificacao       = new WhatsAppService();
        var precoDiaComum     = new PrecoDiaComumStrategy();
        var precoHorarioPico  = new PrecoHorarioPicoStrategy();

        // --- Services ---
        var reservaService = new ReservaService(reservaRepository, notificacao, precoDiaComum, precoHorarioPico);
        var arenaService   = new ArenaService(arenaRepository);

        // --- Controllers ---
        var reservaController = new ReservaController(reservaService);
        var arenaController   = new ArenaController(arenaService);

        // --- View ---
        var view = new ReservaView(reservaController, arenaController);
        view.iniciar();
    }
}
